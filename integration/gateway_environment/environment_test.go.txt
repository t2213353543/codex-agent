type environmentProbe struct {
	accepted atomic.Int64
	active   atomic.Int64
	peak     atomic.Int64
}

func environmentServer(t *testing.T, handler http.HandlerFunc) (*httptest.Server, *environmentProbe) {
	t.Helper()
	probe := &environmentProbe{}
	server := httptest.NewUnstartedServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		active := probe.active.Add(1)
		for old := probe.peak.Load(); active > old && !probe.peak.CompareAndSwap(old, active); old = probe.peak.Load() {
		}
		defer probe.active.Add(-1)
		handler(w, r)
	}))
	server.Config.ConnState = func(_ net.Conn, state http.ConnState) {
		if state == http.StateNew {
			probe.accepted.Add(1)
		}
	}
	server.Start()
	t.Cleanup(server.Close)
	return server, probe
}

func environmentRoute(t *testing.T, server *httptest.Server, id string) ProxyEntry {
	t.Helper()
	u, err := url.Parse(server.URL)
	if err != nil {
		t.Fatal(err)
	}
	port, err := strconv.Atoi(u.Port())
	if err != nil {
		t.Fatal(err)
	}
	return ProxyEntry{ID: id, Name: id, Type: "http", Host: u.Hostname(), Port: port, Enabled: true, Version: 1}
}

func environmentClient(limit int) *http.Client {
	transport := http.DefaultTransport.(*http.Transport).Clone()
	transport.Proxy = nil
	transport.MaxConnsPerHost = limit
	transport.MaxIdleConnsPerHost = limit
	transport.MaxIdleConns = limit * 4
	return &http.Client{Transport: transport, Timeout: 5 * time.Second}
}

func environmentEmit(t *testing.T, data map[string]any) {
	t.Helper()
	data["test"] = t.Name()
	data["synthetic_only"] = true
	raw, err := json.Marshal(data)
	if err != nil {
		t.Fatal(err)
	}
	t.Log("GATEWAY_ENVIRONMENT_METRIC " + string(raw))
}

func environmentWait(t *testing.T, condition func() bool, message string) {
	t.Helper()
	deadline := time.Now().Add(3 * time.Second)
	for time.Now().Before(deadline) {
		if condition() {
			return
		}
		time.Sleep(5 * time.Millisecond)
	}
	t.Fatal(message)
}

func TestGatewayEnvironmentSlowBoundAndReuse(t *testing.T) {
	const requests, limit = 64, 8
	server, probe := environmentServer(t, func(w http.ResponseWriter, r *http.Request) {
		_, _ = io.Copy(io.Discard, r.Body)
		time.Sleep(15 * time.Millisecond)
		_, _ = io.WriteString(w, "synthetic-ok")
	})
	p := &Proxy{Client: environmentClient(limit)}
	t.Cleanup(p.closeGatewayClients)
	route := environmentRoute(t, server, "slow-route")
	var reuse atomic.Int64
	elapsed := make([]float64, requests)
	errorsCh := make(chan error, requests)
	var wg sync.WaitGroup
	start := make(chan struct{})
	for i := range requests {
		wg.Add(1)
		go func(i int) {
			defer wg.Done()
			<-start
			begin := time.Now()
			trace := &httptrace.ClientTrace{GotConn: func(info httptrace.GotConnInfo) {
				if info.Reused {
					reuse.Add(1)
				}
			}}
			ctx := httptrace.WithClientTrace(context.Background(), trace)
			req, _ := http.NewRequestWithContext(ctx, "POST", "http://synthetic.invalid/slow", strings.NewReader("synthetic"))
			res, err := p.clientFor(&route).Do(req)
			if err == nil {
				_, err = io.Copy(io.Discard, res.Body)
				_ = res.Body.Close()
			}
			elapsed[i] = float64(time.Since(begin).Microseconds()) / 1000
			errorsCh <- err
		}(i)
	}
	close(start)
	wg.Wait()
	close(errorsCh)
	for err := range errorsCh {
		if err != nil {
			t.Errorf("local slow request: %v", err)
		}
	}
	sort.Float64s(elapsed)
	p.clientMu.Lock()
	poolSize := len(p.clientPool)
	p.clientMu.Unlock()
	if got := probe.accepted.Load(); got > limit {
		t.Errorf("accepted connections=%d exceed configured limit=%d", got, limit)
	}
	if got := probe.peak.Load(); got > limit {
		t.Errorf("active upstream=%d exceed limit=%d", got, limit)
	}
	if reuse.Load() < requests-limit {
		t.Errorf("reused=%d want at least=%d", reuse.Load(), requests-limit)
	}
	if poolSize != 1 {
		t.Errorf("one route produced %d pool entries", poolSize)
	}
	environmentEmit(t, map[string]any{"requests": requests, "configured_max_connections_per_host": limit, "accepted_connections": probe.accepted.Load(), "peak_upstream_requests": probe.peak.Load(), "reused_requests": reuse.Load(), "pool_entries": poolSize, "p50_ms": elapsed[requests/2-1], "p95_ms": elapsed[60]})
}

func TestGatewayEnvironmentFailuresNoRetry(t *testing.T) {
	const requests = 64
	var calls sync.Map
	server, probe := environmentServer(t, func(w http.ResponseWriter, r *http.Request) {
		_, _ = io.Copy(io.Discard, r.Body)
		id := r.Header.Get("X-Codex-Environment-Case")
		counter, _ := calls.LoadOrStore(id, &atomic.Int32{})
		counter.(*atomic.Int32).Add(1)
		index, _ := strconv.Atoi(id)
		switch index % 4 {
		case 0:
			w.Header().Set("Retry-After", "7")
			w.WriteHeader(429)
		case 1:
			w.WriteHeader(502)
		case 2:
			w.WriteHeader(503)
		case 3:
			connection, _, err := w.(http.Hijacker).Hijack()
			if err != nil {
				t.Errorf("hijack: %v", err)
				return
			}
			if tcp, ok := connection.(*net.TCPConn); ok {
				_ = tcp.SetLinger(0)
			}
			_ = connection.Close()
			return
		}
		_, _ = io.WriteString(w, `{"error":{"message":"synthetic controlled failure"}}`)
	})
	p := newTestProxy(t, requests)
	p.Client = environmentClient(8)
	p.BaseURL = "http://synthetic.invalid"
	route, err := p.Proxies.Create(environmentRoute(t, server, "failure-route"))
	if err != nil {
		t.Fatal(err)
	}
	if err := p.Store.UpdateAccount("test", func(a *Account) error { a.Proxies = []string{route.ID}; return nil }); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(p.closeGatewayClients)
	var wg sync.WaitGroup
	statuses := make([]int, requests)
	start := make(chan struct{})
	begin := time.Now()
	for i := range requests {
		wg.Add(1)
		go func(i int) {
			defer wg.Done()
			<-start
			ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
			defer cancel()
			// Distinct synthetic model scopes keep this transport fault matrix
			// independent from one earlier response cooling down other cases.
			r := proxyRequest("POST", "/v1/responses", fmt.Sprintf(`{"model":"synthetic-%d","stream":true,"input":"synthetic"}`, i)).WithContext(ctx)
			r.Header.Set("X-Codex-Environment-Case", strconv.Itoa(i))
			w := httptest.NewRecorder()
			p.ServeHTTP(w, r)
			statuses[i] = w.Code
			want := []int{429, 502, 503, 502}[i%4]
			if w.Code != want {
				t.Errorf("case=%d HTTP=%d want=%d", i, w.Code, want)
			}
			if i%4 == 0 && w.Header().Get("Retry-After") != "7" {
				t.Errorf("case=%d lost Retry-After", i)
			}
		}(i)
	}
	close(start)
	wg.Wait()
	maxCalls, total := int32(0), int32(0)
	for i := range requests {
		value, ok := calls.Load(strconv.Itoa(i))
		if !ok {
			t.Errorf("request=%d never reached controlled upstream", i)
			continue
		}
		count := value.(*atomic.Int32).Load()
		total += count
		if count > maxCalls {
			maxCalls = count
		}
		if count != 1 {
			t.Errorf("request=%d upstream attempts=%d want=1", i, count)
		}
	}
	account, err := p.Store.Get("test")
	if err != nil {
		t.Fatal(err)
	}
	if p.inFlight(gatewayAdmissionKey(account)) != 0 {
		t.Errorf("account slots leaked: %d", p.inFlight(gatewayAdmissionKey(account)))
	}
	// A new request for the model that already returned 429 must be rejected
	// locally during Retry-After, without replaying it against the upstream.
	followup := proxyRequest("POST", "/v1/responses", `{"model":"synthetic-0","stream":true,"input":"synthetic"}`)
	followup.Header.Set("X-Codex-Environment-Case", "0")
	blocked := httptest.NewRecorder()
	p.ServeHTTP(blocked, followup)
	retryAfter, retryErr := strconv.Atoi(blocked.Header().Get("Retry-After"))
	if blocked.Code != http.StatusTooManyRequests || retryErr != nil || retryAfter < 1 || retryAfter > 7 {
		t.Errorf("cooldown followup HTTP=%d Retry-After=%q", blocked.Code, blocked.Header().Get("Retry-After"))
	}
	if original, ok := calls.Load("0"); !ok || original.(*atomic.Int32).Load() != 1 {
		t.Error("cooldown followup reached upstream")
	}
	distribution := map[int]int{}
	for _, status := range statuses {
		distribution[status]++
	}
	environmentEmit(t, map[string]any{"requests": requests, "upstream_attempts": total, "maximum_attempts_per_request": maxCalls, "client_status_distribution": distribution, "retry_after_preserved": !t.Failed(), "cooldown_followup_blocked_locally": blocked.Code == http.StatusTooManyRequests && !t.Failed(), "peak_upstream_requests": probe.peak.Load(), "duration_ms": float64(time.Since(begin).Microseconds()) / 1000})
}

func TestGatewayEnvironmentCancellationReleasesConnections(t *testing.T) {
	const activeLimit, requests = 4, 20
	server, probe := environmentServer(t, func(w http.ResponseWriter, r *http.Request) {
		_, _ = io.Copy(io.Discard, r.Body)
		if r.URL.Path == "/blocked" {
			select {
			case <-r.Context().Done():
				return
			case <-time.After(5 * time.Second):
				return
			}
		}
		_, _ = io.WriteString(w, "recovered")
	})
	p := &Proxy{Client: environmentClient(activeLimit)}
	t.Cleanup(p.closeGatewayClients)
	route := environmentRoute(t, server, "cancel-route")
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	var wg sync.WaitGroup
	var cancellations atomic.Int64
	for range requests {
		wg.Add(1)
		go func() {
			defer wg.Done()
			req, _ := http.NewRequestWithContext(ctx, "POST", "http://synthetic.invalid/blocked", strings.NewReader("synthetic"))
			res, err := p.clientFor(&route).Do(req)
			if res != nil {
				_ = res.Body.Close()
			}
			if errors.Is(err, context.Canceled) {
				cancellations.Add(1)
			} else {
				t.Errorf("cancel error=%v", err)
			}
		}()
	}
	environmentWait(t, func() bool { return probe.active.Load() == activeLimit }, "configured active connections never saturated")
	begin := time.Now()
	cancel()
	wg.Wait()
	environmentWait(t, func() bool { return probe.active.Load() == 0 }, "cancellation left upstream work active")
	req, _ := http.NewRequest("POST", "http://synthetic.invalid/recover", strings.NewReader("synthetic"))
	res, err := p.clientFor(&route).Do(req)
	if err != nil {
		t.Fatal(err)
	}
	body, err := io.ReadAll(res.Body)
	_ = res.Body.Close()
	if err != nil || string(body) != "recovered" {
		t.Fatalf("connection slot recovery body=%q err=%v", body, err)
	}
	if cancellations.Load() != requests {
		t.Errorf("cancelled=%d want=%d", cancellations.Load(), requests)
	}
	environmentEmit(t, map[string]any{"requests": requests, "configured_max_connections_per_host": activeLimit, "cancelled_requests": cancellations.Load(), "peak_upstream_requests": probe.peak.Load(), "release_and_recovery_ms": float64(time.Since(begin).Microseconds()) / 1000, "followup_request_passed": true})
}

func TestGatewayEnvironmentAccountRouteIsolation(t *testing.T) {
	p := newTestProxy(t, 64)
	p.Client = environmentClient(4)
	var mismatches atomic.Int64
	probes := make([]*environmentProbe, 2)
	accounts := make([]Account, 2)
	for i := range 2 {
		server, probe := environmentServer(t, func(w http.ResponseWriter, r *http.Request) {
			_, _ = io.Copy(io.Discard, r.Body)
			if r.Header.Get("Authorization") != fmt.Sprintf("Bearer synthetic-token-%d", i) || r.Header.Get("ChatGPT-Account-Id") != fmt.Sprintf("synthetic-account-%d", i) {
				mismatches.Add(1)
			}
			_, _ = fmt.Fprintf(w, "route-%d", i)
		})
		probes[i] = probe
		route, err := p.Proxies.Create(environmentRoute(t, server, fmt.Sprintf("route-%d", i)))
		if err != nil {
			t.Fatal(err)
		}
		accounts[i] = Account{Name: fmt.Sprintf("synthetic-account-%d", i), AccessToken: fmt.Sprintf("synthetic-token-%d", i), AccountID: fmt.Sprintf("synthetic-account-%d", i), Proxies: []string{route.ID}}
	}
	t.Cleanup(p.closeGatewayClients)
	var wg sync.WaitGroup
	for n := range 64 {
		wg.Add(1)
		go func(i int) {
			defer wg.Done()
			a := accounts[i]
			route := p.pickProxy(a)
			if route == nil {
				t.Error("account lost route")
				return
			}
			req, _ := http.NewRequest("POST", "http://synthetic.invalid/isolation", strings.NewReader("synthetic"))
			req.Header.Set("Authorization", "Bearer "+a.AccessToken)
			req.Header.Set("ChatGPT-Account-Id", a.AccountID)
			res, err := p.clientFor(route).Do(req)
			if err != nil {
				t.Errorf("route request: %v", err)
				return
			}
			body, err := io.ReadAll(res.Body)
			_ = res.Body.Close()
			if err != nil || string(body) != fmt.Sprintf("route-%d", i) {
				t.Errorf("route=%d body=%q err=%v", i, body, err)
			}
		}(n % 2)
	}
	wg.Wait()
	p.clientMu.Lock()
	size := len(p.clientPool)
	p.clientMu.Unlock()
	if mismatches.Load() != 0 || size != 2 {
		t.Errorf("cross-account header mismatches=%d pool entries=%d", mismatches.Load(), size)
	}
	for i, probe := range probes {
		if probe.accepted.Load() > 4 {
			t.Errorf("route=%d accepted=%d exceed cap", i, probe.accepted.Load())
		}
	}
	environmentEmit(t, map[string]any{"requests": 64, "account_routes": 2, "pool_entries": size, "header_mismatches": mismatches.Load(), "accepted_per_route": []int64{probes[0].accepted.Load(), probes[1].accepted.Load()}})
}

func TestGatewayEnvironmentPoolCapacityAndDefaults(t *testing.T) {
	p := &Proxy{Client: newHTTPClient()}
	t.Cleanup(p.closeGatewayClients)
	for i := range gatewayClientPoolLimit + 17 {
		route := ProxyEntry{ID: fmt.Sprintf("synthetic-route-%d", i), Name: "synthetic", Type: "http", Host: "127.0.0.1", Port: 1, Enabled: true}
		_ = p.clientFor(&route)
	}
	p.clientMu.Lock()
	size := len(p.clientPool)
	p.clientMu.Unlock()
	if size != gatewayClientPoolLimit {
		t.Errorf("pool size=%d want=%d", size, gatewayClientPoolLimit)
	}
	transport := p.Client.Transport.(*http.Transport)
	environmentEmit(t, map[string]any{"created_routes": gatewayClientPoolLimit + 17, "pool_entries": size, "pool_entry_limit": gatewayClientPoolLimit, "default_max_connections_per_host": transport.MaxConnsPerHost, "default_connections_bounded": transport.MaxConnsPerHost > 0, "default_max_idle_connections": transport.MaxIdleConns, "default_max_idle_connections_per_host": transport.MaxIdleConnsPerHost, "note": "Pool metadata bound and live connection bound are different; configured tests do not assert an unlimited product default is bounded."})
}
